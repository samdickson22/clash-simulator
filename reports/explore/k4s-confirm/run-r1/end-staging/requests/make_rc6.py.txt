TAGS = ("pf00", "rd00-a00", "rd00-a01")


def sha(path):
    return hashlib.sha256(open(path, "rb").read()).hexdigest()

s7_hosts, s10_hosts, guard_jobs = {}, {}, {}
for host, r in sorted(roots["hosts"].items()):

                           "guard_job_manifest_sha256": sha(out + "/guard-job-manifest.json"),
                           "bundle": name, "kind": "preflight" if tag.startswith("pf") else "attempt"}
    s7_hosts[host] = {"base": base, "runtime": base + "/" + RUNTIME_LEAF, "bundles": bundles,
                      "repository": base + "/repository", "attempts": r["attempts"], "preflight": r["preflight"],
                      "tools": base + "/tools", "guard_parent": r["guard_job_parent"]}
    s10_hosts[host] = dict(r, runtime_dir=r["runtime"] + "/" + RUNTIME_LEAF,
                           bundle_dirs=[r["bundles"] + "/" + n for n in bundles])

                "print_roots_equals_draft": sha(PR) == sha(DRAFT)},
    "s7_roots_declaration": {"schema": "clasher.k4s.end-roots-declaration.v1",
